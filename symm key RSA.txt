#symm key
import java.util.*;

class SymmetricKey {
static String encrypt(String text, char key) {
String result = "";
for (int i = 0; i < text.length(); i++)
result += (char)(text.charAt(i) ^ key);
return result;
}

public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    System.out.print("Enter plaintext: ");
    String text = sc.nextLine();

    System.out.print("Enter secret key: ");
    char key = sc.nextLine().charAt(0);

    String encrypted = encrypt(text, key);
    System.out.println("Encrypted text: " + encrypted);

    System.out.print("Enter secret key to decrypt: ");
    char key2 = sc.nextLine().charAt(0);

    if (key == key2)
        System.out.println("Decrypted text: " + encrypt(encrypted, key2));
    else
        System.out.println("Enter same key");

    sc.close();
}

}

#RSA
import java.util.;
import java.math.;

class RSA {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);

    System.out.print("Enter prime number p: ");
    int p = sc.nextInt();

    System.out.print("Enter prime number q: ");
    int q = sc.nextInt();

    int n = p * q;
    int phi = (p - 1) * (q - 1);

    System.out.print("Enter public key e: ");
    int e = sc.nextInt();

    BigInteger E = BigInteger.valueOf(e);
    BigInteger N = BigInteger.valueOf(n);
    BigInteger d = E.modInverse(BigInteger.valueOf(phi));

    System.out.print("Enter message (number): ");
    BigInteger msg = sc.nextBigInteger();

    BigInteger enc = msg.modPow(E, N);
    System.out.println("Encrypted message: " + enc);

    BigInteger dec = enc.modPow(d, N);
    System.out.println("Decrypted message: " + dec);

    sc.close();
}

}
