#udp server
import java.net.;
import java.io.;

class UDPServer {
public static void main(String args[]) throws Exception {
DatagramSocket ds = new DatagramSocket(4000);
byte b[] = new byte[1024];

    DatagramPacket dp = new DatagramPacket(b, b.length);
    ds.receive(dp);

    String msg = new String(dp.getData(), 0, dp.getLength());
    System.out.println("Client: " + msg);

    String reply = "Hello Client";
    byte b2[] = reply.getBytes();

    DatagramPacket dp2 = new DatagramPacket(
        b2, b2.length, dp.getAddress(), dp.getPort());
    ds.send(dp2);

    ds.close();
}

}

#udp client
import java.net.;
import java.io.;

class UDPClient {
public static void main(String args[]) throws Exception {
DatagramSocket ds = new DatagramSocket();
InetAddress ip = InetAddress.getLocalHost();

    BufferedReader br = new BufferedReader(
        new InputStreamReader(System.in));

    System.out.print("Enter message: ");
    String msg = br.readLine();
    byte b[] = msg.getBytes();

    DatagramPacket dp = new DatagramPacket(b, b.length, ip, 4000);
    ds.send(dp);

    byte b2[] = new byte[1024];
    DatagramPacket dp2 = new DatagramPacket(b2, b2.length);
    ds.receive(dp2);

    System.out.println("Server: " +
        new String(dp2.getData(), 0, dp2.getLength()));

    ds.close();
}

}

#arp_rapr

import java.util.*;

class ARP_RARP {
public static void main(String[] args) {
Scanner sc = new Scanner(System.in);

    String ip[] = {"192.168.1.1", "192.168.1.2", "192.168.1.3"};
    String mac[] = {"AA:BB:CC:DD:EE:01", "AA:BB:CC:DD:EE:02", "AA:BB:CC:DD:EE:03"};

    System.out.println("1. ARP");
    System.out.println("2. RARP");
    System.out.println("3. Display Full Table");
    System.out.print("Enter choice: ");
    int ch = sc.nextInt();
    sc.nextLine();

    if (ch == 1) {
        System.out.print("Enter IP Address: ");
        String a = sc.nextLine();
        for (int i = 0; i < 3; i++)
            if (ip[i].equals(a))
                System.out.println("MAC Address: " + mac[i]);
    } else if (ch == 2) {
        System.out.print("Enter MAC Address: ");
        String a = sc.nextLine();
        for (int i = 0; i < 3; i++)
            if (mac[i].equalsIgnoreCase(a))
                System.out.println("IP Address: " + ip[i]);
    } else if (ch == 3) {
        System.out.println("IP Address\t\tMAC Address");
        for (int i = 0; i < 3; i++)
            System.out.println(ip[i] + "\t" + mac[i]);
    } else {
        System.out.println("Invalid choice");
    }

    sc.close();
}

}

#time server
import java.net.;
import java.io.;
import java.util.*;

class TimeServer {
public static void main(String[] args) throws Exception {
ServerSocket ss = new ServerSocket(5000);
Socket s = ss.accept();

    PrintWriter out = new PrintWriter(s.getOutputStream(), true);
    out.println(new Date());

    s.close();
    ss.close();
}

}
