#play fair
import java.util.*;

class Playfair {
static char[][] a = new char[5][5];

static void matrix(String key) {
    String s = "";
    key = (key + "abcdefghijklmnopqrstuvwxyz").replace("j", "i");

    for (int i = 0; i < key.length(); i++)
        if (s.indexOf(key.charAt(i)) == -1)
            s += key.charAt(i);

    for (int i = 0, k = 0; i < 5; i++)
        for (int j = 0; j < 5; j++)
            a[i][j] = s.charAt(k++);
}

static String encrypt(String text, int mode) {
    text = text.replace("j", "i").replace(" ", "");
    String s = "";

    for (int i = 0; i < text.length(); i += 2) {
        char x = text.charAt(i);
        char y = (i + 1 < text.length()) ? text.charAt(i + 1) : 'x';

        if (x == y) y = 'x';

        int r1 = 0, c1 = 0, r2 = 0, c2 = 0;

        for (int r = 0; r < 5; r++)
            for (int c = 0; c < 5; c++) {
                if (a[r][c] == x) { r1 = r; c1 = c; }
                if (a[r][c] == y) { r2 = r; c2 = c; }
            }

        if (r1 == r2) {
            s += a[r1][(c1 + mode + 5) % 5];
            s += a[r2][(c2 + mode + 5) % 5];
        } else if (c1 == c2) {
            s += a[(r1 + mode + 5) % 5][c1];
            s += a[(r2 + mode + 5) % 5][c2];
        } else {
            s += a[r1][c2];
            s += a[r2][c1];
        }
    }
    return s;
}

public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    System.out.print("Enter key: ");
    String key = sc.nextLine().toLowerCase();

    matrix(key);

    System.out.print("Enter plaintext: ");
    String text = sc.nextLine().toLowerCase();

    String ct = encrypt(text, 1);
    System.out.println("Encrypted text: " + ct);

    System.out.println("Decrypted text: " + encrypt(ct, -1));

    sc.close();
}

}

import java.util.*;

class Caesar {
static String alpha = "abcdefghijklmnopqrstuvwxyz";

static String encrypt(String msg, int key) {
    String ct = "";
    for (int i = 0; i < msg.length(); i++) {
        int p = alpha.indexOf(msg.charAt(i));
        if (p >= 0)
            ct += alpha.charAt((p + key) % 26);
        else
            ct += msg.charAt(i);
    }
    return ct;
}

static String decrypt(String msg, int key) {
    return encrypt(msg, 26 - key % 26);
}

public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);

    System.out.print("Enter message: ");
    String msg = sc.nextLine().toLowerCase();

    System.out.print("Enter shift key: ");
    int key = sc.nextInt();

    String ct = encrypt(msg, key);
    System.out.println("Encrypted message: " + ct);

    String pt = decrypt(ct, key);
    System.out.println("Decrypted message: " + pt);

    sc.close();
}
